function createColumnGroups(regionId) {

    const region = document.getElementById(regionId);

    console.log(region)

    if (!region) {
        return;
    }

    //const table = region.querySelector("table");
    const table = document.getElementById("report_table_" + regionId);

  
    console.log(table)

    if (!table) {
        return;
    }

    const thead = table.querySelector("thead");

    console.log(thead)

    if (!thead) {
        return;
    }

    /*
     * Remove previously generated group row
     * so this also works after report refresh.
     */
    const existingGroupRow = thead.querySelector(".dynamic-column-group-row");

    if (existingGroupRow) {
        existingGroupRow.remove();
    }

    /*
     * Get the normal APEX column heading row.
     */
    const headerRow = thead.querySelector("tr");

    console.log('header' + headerRow)

    if (!headerRow) {
        return;
    }

    const headers = Array.from(headerRow.querySelectorAll("th"));

    console.log('thead' + headers)

    if (!headers.length) {
        return;
    }

    /*
     * Create the new group row.
     */
    const groupRow = document.createElement("tr");

    groupRow.className = "dynamic-column-group-row";

    let currentGroup = null;
    let currentGroupCell = null;

    headers.forEach(function (th) {

        /*
         * Get the displayed heading.
         *
         * Example:
         * Sales|Jan
         * Sales|Feb
         * Expense|Jan
         */
        const headingText = th.textContent.trim();

        const separatorIndex = headingText.indexOf("|");

        /*
         * No "|" means this is an ungrouped column.
         */
        if (separatorIndex === -1) {

            const groupCell = document.createElement("th");

            groupCell.className = "t-Report-colHead";

            groupCell.textContent = "";

            groupRow.appendChild(groupCell);

            return;
        }

        const groupName =
            headingText.substring(0, separatorIndex).trim();

        const columnName =
            headingText.substring(separatorIndex + 1).trim();

        /*
         * Replace the displayed heading with only
         * the child column name.
         */
        const headingElement = th.querySelector("a, span");

        if (headingElement) {

            headingElement.textContent = columnName;

        } else {

            th.textContent = columnName;

        }

        /*
         * Same group as previous column.
         */
        if (currentGroup === groupName) {

            currentGroupCell.colSpan++;

        }

        /*
         * New group.
         */
        else {

            currentGroup = groupName;

            currentGroupCell = document.createElement("th");

            currentGroupCell.className = "t-Report-colHead";

            currentGroupCell.textContent = groupName;

            currentGroupCell.colSpan = 1;

            groupRow.appendChild(currentGroupCell);
        }
    });

    /*
     * Insert group row above the normal APEX header row.
     */
    thead.insertBefore(groupRow, headerRow);

    console.log('Inserted')
}
