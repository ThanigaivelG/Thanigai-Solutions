CREATE TABLE PORTAL_MENU_T
(
    MENU_ID          NUMBER NOT NULL,
    MENU_NAME        VARCHAR2(100 CHAR) NOT NULL,
    PARENT_ID        NUMBER,
    APPLICATION_ID   NUMBER,
    DISPLAY_SEQUENCE NUMBER,
    PAGE_NUMBER      NUMBER,
    ICON_NAME        VARCHAR2(50 CHAR),
    IMAGE_PATH       VARCHAR2(50 CHAR),
    DESCRIPTION      VARCHAR2(4000 CHAR),
    CREATED_BY       VARCHAR2(255 CHAR),
    CREATED_ON       TIMESTAMP(6) DEFAULT SYSTIMESTAMP,
    UPDATED_BY       VARCHAR2(255 CHAR),
    UPDATED_ON       TIMESTAMP(6),
    STATUS           CHAR(1) DEFAULT 'Y' NOT NULL,

    CONSTRAINT PORTAL_MENU_PK
        PRIMARY KEY (MENU_ID),

    CONSTRAINT PORTAL_MENU_STATUS_CK
        CHECK (STATUS IN ('Y', 'N'))
);

select level,
       MENU_NAME as label, 
       'f?p=&APP_ID.:' || PAGE_NUMBER||':&SESSION.:::::' TARGET,
       CASE
        WHEN TO_NUMBER(PAGE_NUMBER) =  :APP_PAGE_ID THEN 'YES' 
        ELSE 'NO'
        END AS is_current,
        ICON_NAME as icon,   
       case when connect_by_isleaf = 1 then 0
            when level = 1             then 1
            else                           -1
       end as parent_id
FROM PORTAL_MENU_T
WHERE APPLICATION_ID = 100
  AND STATUS = 'Y' 
START WITH PARENT_ID IS NULL 
CONNECT BY PRIOR MENU_ID = PARENT_ID 
ORDER SIBLINGS BY DISPLAY_SEQUENCE;
